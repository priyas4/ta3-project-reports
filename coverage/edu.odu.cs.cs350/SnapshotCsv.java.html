package edu.odu.cs.cs350;

import java.io.IOException;
import java.io.Reader;
import java.util.ArrayList;
import java.util.List;
import java.util.Set;
import org.apache.commons.csv.CSVFormat;
import org.apache.commons.csv.CSVRecord;

/** Loads lecture observations from a single enrollment snapshot. */
public final class SnapshotCsv {
    /** Columns required for a course summary. */
    private static final List<String> HEADERS = List.of(
            "CRN", "SUBJ", "CRSE", "LINK", "XLST GROUP", "OVERALL ENR", "OVERALL CAP");

    private SnapshotCsv() {}

    /** Reads a headered CSV using the supplied exact lecture LINK values.
     * @param input reader for the snapshot
     * @param lectureLinks accepted lecture LINK values
     * @return lecture observations from the snapshot
     * @throws IOException if the snapshot cannot be read or is malformed
     */
    public static List<Section> read(Reader input, Set<String> lectureLinks) throws IOException {
        var format = CSVFormat.DEFAULT.builder().setHeader().setSkipHeaderRecord(true)
                .setTrim(true).build();
        try (var rows = format.parse(input)) {
            for (String header : HEADERS) {
                if (!rows.getHeaderMap().containsKey(header)) {
                    throw new IOException("Missing CSV column: " + header);
                }
            }
            var sections = new ArrayList<Section>();
            for (CSVRecord row : rows.getRecords()) {
                if (!row.isConsistent()) {
                    throw new IOException("CSV record " + row.getRecordNumber() + " has the wrong number of columns");
                }
                if (lectureLinks.contains(row.get("LINK"))) {
                    sections.add(lecture(row));
                }
            }
            return sections;
        }
    }

    private static Section lecture(CSVRecord row) throws IOException {
        try {
            String course = required(row, "SUBJ") + required(row, "CRSE");
            String group = row.get("XLST GROUP");
            String offering = group.isEmpty() ? "crn:" + required(row, "CRN") : "group:" + group;
            int enrolled = Integer.parseInt(row.get("OVERALL ENR"));
            int capacity = Integer.parseInt(row.get("OVERALL CAP"));
            if (enrolled < 0 || capacity < 0) {
                throw new IllegalArgumentException("Enrollment and capacity must be nonnegative");
            }
            return new Section(course, offering, true, enrolled, capacity);
        } catch (IllegalArgumentException ex) {
            throw new IOException("CSV record " + row.getRecordNumber() + ": " + ex.getMessage(), ex);
        }
    }

    private static String required(CSVRecord row, String column) {
        String value = row.get(column);
        if (value.isEmpty()) {
            throw new IllegalArgumentException("Missing " + column);
        }
        return value;
    }
}
