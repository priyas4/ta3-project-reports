package edu.odu.cs.cs350;

import java.io.IOException;
import java.io.PrintStream;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDate;
import java.time.format.DateTimeParseException;
import java.util.Arrays;
import java.util.HashSet;
import java.util.List;

/** Runs the Sprint 1 enrollment summary using sample records or a CSV snapshot. */
public final class SummaryDemo {
    private SummaryDemo() {}

    /** Prints the sample report or summarizes a selected or dated CSV snapshot.
     * @param args command-line options and input paths
     */
    public static void main(String[] args) {
        int status = run(args, System.out, System.err);
        if (status != 0) {
            System.exit(status);
        }
    }

    /** Runs the command without exiting the caller's process.
     * @param args command-line options and paths
     * @param output destination for the report
     * @param errors destination for error messages
     * @return zero on success, or one for invalid arguments or input
     */
    static int run(String[] args, PrintStream output, PrintStream errors) {
        if (args.length != 0) {
            try {
                printFromInput(args, output);
            } catch (IOException | IllegalArgumentException | DateTimeParseException ex) {
                errors.println(ex.getMessage());
                return 1;
            }
            return 0;
        }
        var sections = List.of(
                new Section("CS350", "A", true, 42, 50),
                new Section("CS350", "A", true, 42, 50),
                new Section("CS350", "B", true, 20, 25),
                new Section("CS350", "Lab", false, 30, 30),
                new Section("CS150", "A", true, 10, 15));
        output.print(SummaryReport.render(sections));
        return 0;
    }

    private static void printFromInput(String[] args, PrintStream output) throws IOException {
        Path snapshot;
        int linksStart;
        if (args.length >= 3 && args[0].equals("--csv")) {
            snapshot = Path.of(args[1]);
            linksStart = 2;
        } else if (args.length >= 5 && args[0].equals("--semester")) {
            snapshot = SemesterSnapshots.latest(Path.of(args[1]),
                    LocalDate.parse(args[2]), LocalDate.parse(args[3]));
            linksStart = 4;
            output.println("Snapshot: " + snapshot.getFileName());
        } else {
            throw new IllegalArgumentException("Usage: --csv FILE LINK [LINK ...]"
                    + " or --semester DIR START CUTOFF LINK [LINK ...]");
        }
        try (var input = Files.newBufferedReader(snapshot)) {
            var lectureLinks = new HashSet<>(Arrays.asList(args).subList(linksStart, args.length));
            output.print(SummaryReport.render(SnapshotCsv.read(input, lectureLinks)));
        }
    }
}
