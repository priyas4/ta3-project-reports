package edu.odu.cs.cs350;

import java.io.IOException;
import java.nio.file.Files;
import java.nio.file.Path;
import java.time.LocalDate;
import java.time.format.DateTimeParseException;
import java.util.Comparator;
import java.util.Optional;

/** Finds the most recent eligible enrollment snapshot in a semester directory. */
public final class SemesterSnapshots {
    private SemesterSnapshots() {}

    /** Selects the newest dated CSV between the supplied dates, inclusive.
     * @param directory semester data directory
     * @param start first eligible date
     * @param cutoff last eligible date
     * @return path to the newest eligible CSV
     * @throws IOException if the directory cannot be read or has no eligible file
     */
    public static Path latest(Path directory, LocalDate start, LocalDate cutoff) throws IOException {
        if (cutoff.isBefore(start)) {
            throw new IllegalArgumentException("Cutoff cannot precede registration start");
        }
        try (var files = Files.list(directory)) {
            return files.filter(Files::isRegularFile)
                    .map(SemesterSnapshots::datedSnapshot)
                    .flatMap(Optional::stream)
                    .filter(snapshot -> !snapshot.date().isBefore(start)
                            && !snapshot.date().isAfter(cutoff))
                    .max(Comparator.comparing(DatedSnapshot::date))
                    .orElseThrow(() -> new IOException("No eligible dated CSV in " + directory))
                    .file();
        }
    }

    private static Optional<DatedSnapshot> datedSnapshot(Path file) {
        String name = file.getFileName().toString();
        if (!name.matches("[0-9]{4}-[0-9]{2}-[0-9]{2}\\.csv")) {
            return Optional.empty();
        }
        try {
            return Optional.of(new DatedSnapshot(file, LocalDate.parse(name.substring(0, 10))));
        } catch (DateTimeParseException ex) {
            return Optional.empty();
        }
    }

    private record DatedSnapshot(Path file, LocalDate date) {}
}
