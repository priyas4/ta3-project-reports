package edu.odu.cs.cs350;

import java.util.HashMap;
import java.util.Map;
import java.util.TreeMap;

/** Formats current lecture enrollment and capacity by course. */
public final class SummaryReport {
    private SummaryReport() {}

    /** Counts each offering once within its course and prints courses in order.
     * @param sections input section observations
     * @return the formatted course summary
     */
    public static String render(Iterable<Section> sections) {
        var courses = new TreeMap<String, Map<String, Section>>();
        for (Section section : sections) {
            if (section.lecture()) {
                var offerings = courses.computeIfAbsent(section.course(), key -> new HashMap<>());
                var existing = offerings.putIfAbsent(section.offering(), section);
                if (existing != null && (existing.enrollment() != section.enrollment()
                        || existing.capacity() != section.capacity())) {
                    throw new IllegalArgumentException("Conflicting totals for " + section.course()
                            + " offering " + section.offering());
                }
            }
        }
        var output = new StringBuilder("Course\tEnrolled\tCapacity\n");
        courses.forEach((course, offerings) -> {
            int enrolled = offerings.values().stream().mapToInt(Section::enrollment).sum();
            int capacity = offerings.values().stream().mapToInt(Section::capacity).sum();
            output.append(course).append('\t').append(enrolled).append('\t').append(capacity).append('\n');
        });
        return output.toString();
    }
}
